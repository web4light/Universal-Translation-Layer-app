█ KROK 1: INSTALACE GNAT/SPARK
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

sudo apt update
sudo apt install -y build-essential python3 python3-pip

# Stáhnout GNAT Community (nebo použít AdaDev2024.zip)
# https://www.adacore.com/community

# Po instalaci:
export PATH="/opt/GNAT/2024/bin:$PATH"
echo 'export PATH="/opt/GNAT/2024/bin:$PATH"' >> ~/.bashrc


█ KROK 2: PYTHON DEPENDENCIES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

pip3 install prometheus_client


█ KROK 3: ROZBALIT A SPUSTIT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# Rozbal balíček
tar -xzf vakuova-mincovna-v1.0.tar.gz
# nebo
unzip vakuova-mincovna-v1.0.zip

# Přejdi do složky
cd vakuova-mincovna

# Dej práva
chmod +x start.sh

# SPUSŤ!
./start.sh
